cmake_minimum_required(VERSION 3.28)
project(NekoTuneKugou LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Network)
qt_standard_project_setup(REQUIRES 6.6)
set(CMAKE_AUTOMOC ON)
include(CTest)
set(NEKOTUNE_SDK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.nekotune-sdk/sdk" CACHE PATH "SDK source directory")
find_package(NekoTuneSDK 2 QUIET)
if(NOT NekoTuneSDK_FOUND)
    if(NOT EXISTS "${NEKOTUNE_SDK_SOURCE_DIR}/CMakeLists.txt")
        message(FATAL_ERROR "Set CMAKE_PREFIX_PATH to an installed SDK, or NEKOTUNE_SDK_SOURCE_DIR to the host's sdk directory. See README.md.")
    endif()
    add_subdirectory("${NEKOTUNE_SDK_SOURCE_DIR}" nekotune-sdk)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/nekotune-sdk.json" sdk_info)
string(JSON expected_qt GET "${sdk_info}" qt)
string(REGEX MATCH "^[0-9]+\\.[0-9]+" expected_qt_minor "${expected_qt}")
if(NOT "${Qt6_VERSION_MAJOR}.${Qt6_VERSION_MINOR}" STREQUAL expected_qt_minor)
    message(FATAL_ERROR "This release targets Qt ${expected_qt_minor}.x; found ${Qt6_VERSION}. Update nekotune-sdk.json when upgrading the host baseline.")
endif()
add_library(extension_common STATIC common/http.cpp common/json.cpp common/media.cpp)
target_include_directories(extension_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(extension_common PUBLIC NekoTune::SDK Qt6::Core Qt6::Network)
set_target_properties(extension_common PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
find_package(ZLIB REQUIRED)
add_library(extension_core STATIC kugou/client.cpp kugou/extension.cpp kugou/lyrics.cpp)
target_link_libraries(extension_core PUBLIC extension_common ZLIB::ZLIB)
set_target_properties(extension_core PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
file(GLOB_RECURSE plugin_qml CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/kugou/qml/*")
nekotune_add_extension(nekotune_kugou
    MANIFEST kugou/extension.json SOURCES kugou/plugin.cpp FILES ${plugin_qml})
target_link_libraries(nekotune_kugou PRIVATE extension_core)
add_custom_target(extension_package ALL DEPENDS nekotune_kugou_package)
_nekotune_platform_key(build_platform)
configure_file(cmake/build-info.json.in "${CMAKE_BINARY_DIR}/build-info.json" @ONLY)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test Qml)
    add_executable(kugou_test tests/unit/kugou_test.cpp)
    target_link_libraries(kugou_test PRIVATE extension_core Qt6::Test)
    add_test(NAME protocol COMMAND kugou_test)
    add_executable(translations_test tests/unit/translations_test.cpp)
    target_link_libraries(translations_test PRIVATE Qt6::Qml Qt6::Test)
    target_compile_definitions(translations_test PRIVATE NEKOTUNE_EXTENSIONS_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME translations COMMAND translations_test)
    add_executable(plugin_lifecycle_test tests/plugin_lifecycle_test.cpp)
    target_include_directories(plugin_lifecycle_test PRIVATE tests/unit)
    target_link_libraries(plugin_lifecycle_test PRIVATE NekoTune::SDK Qt6::Test)
    add_dependencies(plugin_lifecycle_test nekotune_kugou)
    add_test(NAME plugin_lifecycle COMMAND plugin_lifecycle_test
        $<TARGET_FILE:nekotune_kugou> "${CMAKE_CURRENT_SOURCE_DIR}/kugou/extension.json")
endif()
